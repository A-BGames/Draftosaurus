CREATE DATABASE draftosaurus;
USE draftosaurus;

CREATE TABLE Jugador (
    id_jugador INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL
);

CREATE TABLE Partida (
    id_partida INT PRIMARY KEY AUTO_INCREMENT,
    fecha DATE NOT NULL,
    modo_de_juego ENUM('Estándar', 'Solo', 'Invierno') NOT NULL,
    ganador INT,
    FOREIGN KEY (ganador) REFERENCES Jugador(id_jugador)
);

CREATE TABLE Tablero (
    id_tablero INT PRIMARY KEY AUTO_INCREMENT,
    id_jugador INT UNIQUE NOT NULL,
    FOREIGN KEY (id_jugador) REFERENCES Jugador(id_jugador)
);
CREATE TABLE Dinosaurio (
    id_dinosaurio INT  AUTO_INCREMENT,
    id_tablero    INT  NOT NULL,
    id_jugador    INT  NOT NULL,
    especie VARCHAR(50) NOT NULL,
    PRIMARY KEY (id_dinosaurio,id_jugador,id_tablero),
    FOREIGN KEY (id_jugador) REFERENCES Jugador(id_jugador),
    FOREIGN KEY (id_tablero) REFERENCES Tablero(id_tablero)	
);

CREATE TABLE Zona_Tablero (
    id_zona INT PRIMARY KEY AUTO_INCREMENT,
    id_tablero INT NOT NULL,
    nombre_zona VARCHAR(50) NOT NULL,
    regla_puntuacion TEXT NOT NULL,
    FOREIGN KEY (id_tablero) REFERENCES Tablero(id_tablero)
);


CREATE TABLE Ronda (
    id_ronda INT  AUTO_INCREMENT,
    id_partida INT NOT NULL,
    numero_ronda INT CHECK (numero_ronda IN (1, 2)),
    PRIMARY KEY (id_ronda, id_partida), 
    FOREIGN KEY (id_partida) REFERENCES partida(id_partida) 
);

CREATE TABLE Turno (
    id_turno INT PRIMARY KEY AUTO_INCREMENT,
    numero_turno INT CHECK (numero_turno BETWEEN 1 AND 6),
    id_ronda INT,
    FOREIGN KEY (id_ronda) REFERENCES Ronda(id_ronda)
);
CREATE TABLE partida_jugador (
    id_partida INT NOT NULL, 
    id_jugador INT NOT NULL,
    puntuacion INT NOT NULL,

    PRIMARY KEY (id_partida, id_jugador), 
    FOREIGN KEY (id_partida) REFERENCES partida(id_partida), 
    FOREIGN KEY (id_jugador) REFERENCES jugador(id_jugador)  
);

CREATE TABLE jugador_tablero (
    id_jugador INT NOT NULL,
    id_tablero INT NOT NULL,

    PRIMARY KEY (id_tablero, id_jugador), 
    FOREIGN KEY (id_tablero) REFERENCES tablero(id_tablero), 
    FOREIGN KEY (id_jugador) REFERENCES jugador(id_jugador)  
);
CREATE TABLE ronda_jugador (
    id_jugador INT NOT NULL,
    id_ronda INT NOT NULL,
    id_turno INT NOT NULL,

    PRIMARY KEY (id_ronda, id_jugador,id_turno), 
    FOREIGN KEY (id_ronda) REFERENCES ronda(id_ronda), 
    FOREIGN KEY (id_jugador) REFERENCES jugador(id_jugador),  
    FOREIGN KEY (id_turno) REFERENCES turno(id_turno)
);
CREATE TABLE dinosaurio_por_zona(
    id_dinosaurio INT NOT NULL,
    id_zona       INT NOT NULL,
    zona_puntos   INT NOT NULL,
    PRIMARY KEY (id_dinosaurio,id_zona),
    FOREIGN KEY (id_dinosaurio) REFERENCES dinosaurio(id_dinosaurio),
    FOREIGN KEY (id_zona) REFERENCES zona_tablero(id_zona)
);

--Consultas Basicas
SELECT j.nombre
FROM partida_jugador pj
INNER JOIN jugador j ON pj.id_jugador = j.id_jugador
WHERE pj.id_partida = 1;


SELECT 
  pj.id_partida,
  pj.id_jugador,
  d.id_dinosaurio,
  d.especie,
  dpz.id_zona
FROM partida_jugador pj
INNER JOIN dinosaurio d 
  ON pj.id_jugador = d.id_jugador
INNER JOIN dinosaurio_por_zona dpz 
  ON d.id_dinosaurio = dpz.id_dinosaurio;

SELECT 
  d.id_jugador,
  pj.id_partida,
  SUM(dpz.zona_puntos) AS puntuacion_total
FROM dinosaurio_por_zona dpz
INNER JOIN dinosaurio d ON dpz.id_dinosaurio = d.id_dinosaurio
INNER JOIN partida_jugador pj ON d.id_jugador = pj.id_jugador
GROUP BY d.id_jugador, pj.id_partida;

SELECT 
    d.id_jugador,
    j.nombre AS nombre_jugador,
    pj.id_partida,
    SUM(dpz.zona_puntos) AS puntuacion_total,
    SUM(CASE WHEN d.especie = 'Tiranosaurio Rex' THEN 1 ELSE 0 END) AS cantidad_trex
FROM dinosaurio d
INNER JOIN dinosaurio_por_zona dpz ON d.id_dinosaurio = dpz.id_dinosaurio
INNER JOIN partida_jugador pj ON d.id_jugador = pj.id_jugador
INNER JOIN jugador j ON d.id_jugador = j.id_jugador
GROUP BY d.id_jugador, pj.id_partida, j.nombre
ORDER BY puntuacion_total DESC, cantidad_trex DESC
LIMIT 1;
